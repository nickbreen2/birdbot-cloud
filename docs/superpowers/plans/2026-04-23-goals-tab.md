## Task 1: GET /api/goals route

**Files:**
- Create: `src/app/api/goals/route.ts`

- [ ] **Step 1: Create the route file**

```ts
import { NextResponse } from "next/server"
import { auth } from "@clerk/nextjs/server"
import { prisma } from "@/lib/prisma"

export async function GET() {
  try {
    const { userId } = await auth()
    if (!userId) {
      return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
    }

    const goals = await prisma.goal.findMany({
      where: { userId },
      orderBy: { createdAt: "desc" },
      select: {
        id: true,
        title: true,
        description: true,
        status: true,
        cadence: true,
        lastCheckinAt: true,
        targetDate: true,
        createdAt: true,
        entityId: true,
        entity: {
          select: {
            mentions: {
              orderBy: { createdAt: "asc" },
              take: 1,
              select: { snippet: true },
            },
          },
        },
      },
    })

    const pending = goals
      .filter((g) => g.status === "pending_confirmation")
      .map((g) => ({
        id: g.id,
        title: g.title,
        description: g.description,
        createdAt: g.createdAt,
        entityId: g.entityId,
        triggerSnippet: g.entity.mentions[0]?.snippet ?? null,
      }))

    const active = goals
      .filter((g) => g.status === "active")
      .map((g) => ({
        id: g.id,
        title: g.title,
        description: g.description,
        cadence: g.cadence,
        lastCheckinAt: g.lastCheckinAt,
        targetDate: g.targetDate,
        createdAt: g.createdAt,
        entityId: g.entityId,
      }))

    return NextResponse.json({ pending, active })
  } catch (err) {
    console.error("[GET /api/goals] Error:", err)
    const message = err instanceof Error ? err.message : "Unknown error"
    return NextResponse.json({ error: message }, { status: 500 })
  }
}
```

## Task 2: PATCH /api/goals/[id] route

**Files:**
- Create: `src/app/api/goals/[id]/route.ts`

- [ ] **Step 1: Create the route file**

```ts
import { NextRequest, NextResponse } from "next/server"
import { auth } from "@clerk/nextjs/server"
import { prisma } from "@/lib/prisma"
import { GoalStatus } from "@/generated/prisma/enums"

const VALID_STATUSES = new Set<string>([
  "active",
  "paused",
  "completed",
  "abandoned",
])

export async function PATCH(
  request: NextRequest,
  { params }: { params: Promise<{ id: string }> }
) {
  try {
    const { userId } = await auth()
    if (!userId) {
      return NextResponse.json({ error: "Unauthorized" }, { status: 401 })
    }

    const { id } = await params
    const body = await request.json()
    const { status } = body as { status: string }

    if (!status || !VALID_STATUSES.has(status)) {
      return NextResponse.json(
        { error: "Invalid status. Must be one of: active, paused, completed, abandoned" },
        { status: 400 }
      )
    }

    const existing = await prisma.goal.findFirst({ where: { id, userId } })
    if (!existing) {
      return NextResponse.json({ error: "Not found" }, { status: 404 })
    }

    const updated = await prisma.goal.update({
      where: { id },
      data: { status: status as GoalStatus },
      select: {
        id: true,
        title: true,
        description: true,
        status: true,
        cadence: true,
        lastCheckinAt: true,
        targetDate: true,
        createdAt: true,
        entityId: true,
      },
    })

    return NextResponse.json(updated)
  } catch (err) {
    console.error("[PATCH /api/goals/[id]] Error:", err)
    const message = err instanceof Error ? err.message : "Unknown error"
    return NextResponse.json({ error: message }, { status: 500 })
  }
}
```
